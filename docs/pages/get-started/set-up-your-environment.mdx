from flask import Flask, request, redirect, render_template_string
from flask_sqlalchemy import SQLAlchemy
import stripe
import requests
from bs4 import BeautifulSoup
import ollama

# ====================================================
# CONFIG
# ====================================================

app = Flask(__name__)

app.config["SQLALCHEMY_DATABASE_URI"] = "sqlite:///shop.db"

db = SQLAlchemy(app)

# STRIPE
stripe.api_key = "TA_CLE_STRIPE"

# ====================================================
# DATABASE
# ====================================================

class Product(db.Model):

    id = db.Column(db.Integer, primary_key=True)

    name = db.Column(db.String(200))

    description = db.Column(db.Text)

    price = db.Column(db.Float)

    image = db.Column(db.String(1000))

    supplier_url = db.Column(db.String(1000))

# ====================================================
# AI DESCRIPTION
# ====================================================

def generate_description(product_name):

    prompt = f"""
Crée une fiche produit ecommerce optimisée SEO.

Produit :
{product_name}

Ajoute :
- titre vendeur
- bénéfices
- description persuasive
"""

    response = ollama.chat(

        model="llama3",

        messages=[
            {
                "role": "user",
                "content": prompt
            }
        ]
    )

    return response["message"]["content"]

# ====================================================
# IMPORT PRODUIT URL
# ====================================================

def import_product(url):

    headers = {
        "User-Agent": "Mozilla/5.0"
    }

    html = requests.get(
        url,
        headers=headers
    ).text

    soup = BeautifulSoup(html, "html.parser")

    title = soup.title.text[:100]

    image = ""

    for img in soup.find_all("img"):

        src = img.get("src")

        if src and "http" in src:

            image = src
            break

    description = generate_description(title)

    product = Product(

        name=title,

        description=description,

        price=49.99,

        image=image,

        supplier_url=url
    )

    db.session.add(product)

    db.session.commit()

@app.route("/add", methods=["GET", "POST"])
def add():

    if request.method == "POST":

        url = request.form["url"]

        import_product(url)

        return redirect("/")

    return """

    <h1>Importer Produit</h1>

    <form method="POST">

        <input
        name="url"
        placeholder="URL fournisseur">

        <button>Importer</button>

    </form>

@app.route("/buy/<int:id>")
def buy(id):

    product = Product.query.get(id)

    session = stripe.checkout.Session.create(

        payment_method_types=["card"],

        line_items=[{

            "price_data": {

                "currency": "eur",

                "product_data": {
                    "name": product.name
                },

                "unit_amount": int(
                    product.price * 100
                ),
            },

            "quantity": 1,
        }],

        mode="payment",

        success_url=f"http://localhost:5000/success/{id}",

        cancel_url="http://localhost:5000",
    )

    return redirect(session.url)
